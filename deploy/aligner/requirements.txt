# TOP-LEVEL PINS ONLY, NOT a transitive lock and NOT hash-pinned: each
# package this image imports directly is pinned here, and pip resolves their
# transitive dependencies at build time. requirements.in lists the same set.
#
# deploy/yamnet-detector/requirements.txt is a full transitive lock with
# hashes (`uv pip compile --generate-hashes`, installed with
# `--require-hashes`). Doing the same here means resolving torch (CPU index)
# and demucs's whole closure, several GB of wheels; that lock, and switching
# the Dockerfile to --require-hashes, is issue #1017.
fastapi==0.138.0
uvicorn[standard]==0.49.0
python-multipart==0.0.32
# torch/torchaudio held at 2.8.0: torchaudio 2.9 moves load/save onto
# TorchCodec (its matching 0.8.x has no linux/arm64 wheel), and removes
# list_audio_backends and torchaudio.functional.forced_align/merge_tokens,
# which this sidecar's own forced-alignment code calls directly (#1016
# dropped whisperx, which is the only thing that called list_audio_backends;
# forced_align/merge_tokens are still this sidecar's own dependency). 2.8.0
# ships cp311 manylinux wheels for both aarch64 and x86_64.
# The Dockerfile installs these two from the PyTorch CPU index
# (https://download.pytorch.org/whl/cpu) BEFORE this file, reading the
# versions from the lines below: PyPI's x86_64 torch is the CUDA build and
# pulls ~5 GB of nvidia-*-cu12 libraries. Keep the pins in exactly this
# `name==version` form -- the Dockerfile greps for them and fails the build
# if either is missing.
torch==2.8.0
torchaudio==2.8.0
demucs==4.0.1
# faster-whisper: the transcript pass (#1016, replaces whisperx, which only
# wrapped this same library plus its own bundled pyannote VAD). transformers:
# the Hugging Face wav2vec2 forced-alignment models for the non-torchaudio
# languages (see app.ALIGN_MODELS / _aligner_models.FasterWhisperAligner).
# Pinned at 5.17.0/1.2.1, the same versions the whisperx image's own freeze
# carried (measured, #1016 review). A fresh resolution of this file first
# picked transformers 4.57.1, which carries known advisories a current
# release fixes (PYSEC-2026-2289, a config.json-driven RCE fixed in 5.3.0;
# PYSEC-2026-3929, fixed in 5.10.0), and _load_align_model `from_pretrained`s
# third-party, non-revision-pinned HF repo ids (jonatasgrosman/*, Yehor/*,
# ...) from app.ALIGN_MODELS -- exactly PYSEC-2026-2289's vector. Verified
# compatible: bit-identical wav2vec2 logits on identical input between
# transformers 4.57.1 and 5.17.0 (#1016 hostile review).
faster-whisper==1.2.1
transformers==5.17.0
